/** Printable lab report helper — Phase 6.1 enhanced version */

import { useI18nStore } from '../stores/i18n.store'
import { useAuthStore }    from '../stores/auth'
import { useStudentStore } from '../stores/student.store'

/* ─── Option types ─── */

export interface LabReportTable {
  caption?: string
  headers: (string | number)[]
  rows: (string | number | undefined)[][]
}

export interface LabReportImage {
  src: string
  caption?: string
  fullWidth?: boolean
}

export interface LabReportHtmlBlock {
  title?: string
  html: string
}

/** A single experiment parameter shown in the params section */
export interface LabReportParam {
  label: string
  value: string | number
  unit?: string
}

/** A key result shown in the summary card row */
export interface LabReportStat {
  label: string
  value: string | number
  unit?: string
  highlight?: boolean   // orange colour
}

export interface OpenLabReportOptions {
  title: string
  icon?: string               // emoji icon, e.g. '🧲'
  experimentName?: string
  dir?: 'rtl' | 'ltr'
  dateLocale?: string
  meta?: Record<string, string>

  /** Experiment parameters — auto-rendered as a formatted two-column table */
  params?: LabReportParam[]

  /** Key results — shown as styled summary cards below the params */
  summaryStats?: LabReportStat[]

  tables?: LabReportTable[]
  images?: LabReportImage[]
  htmlBlocks?: LabReportHtmlBlock[]
  footerNote?: string

  /** If provided, auto-inserted as the first image with caption */
  canvasSnapshot?: string

  /** Open the browser print dialog after render (Save as PDF) */
  openPrintDialog?: boolean

  /** If true, show "Send to Teacher" button (guest mode) */
  sendToTeacher?: boolean
}

/* ─── Main function ─── */

export function openLabReport(options: OpenLabReportOptions): Window | null {
  const i18nStore = useI18nStore()
  const dir   = options.dir ?? i18nStore.direction ?? 'rtl'
  const locale = options.dateLocale ?? i18nStore.locale ?? 'ar'
  const t     = i18nStore.t

  const strings = {
    dateLabel:       t('labReport.meta.date'),
    experimentLabel: t('labReport.meta.experiment'),
    paramsHeading:   t('labReport.paramsHeading'),
    resultsHeading:  t('labReport.resultsHeading'),
    footerHint:      t('labReport.footerHint'),
    printLabel:      t('labReport.actions.print'),
    closeLabel:      t('labReport.actions.close'),
    simulationSnap:  t('labReport.simulationSnapshot'),
  }

  const now = new Date().toLocaleString(locale)

  // Auto-enable send-to-teacher for guest users
  const isGuest = (() => {
    try {
      const auth = useAuthStore()
      return auth.isGuest
    } catch { return false }
  })()
  const sendToTeacherScript = (options.sendToTeacher || isGuest) ? buildSendScript(options) : ''

  /* ── student info section ── */
  let studentHtml = ''
  try {
    const auth    = useAuthStore()
    const student = useStudentStore()
    const rows: [string, string][] = []
    if (auth.user?.name)        rows.push([dir === 'rtl' ? 'الاسم' : 'Name',              auth.user.name])
    if (student.studentId)  rows.push([dir === 'rtl' ? 'رقم الطالب' : 'Student ID',  student.studentId])
    if (student.course)     rows.push([dir === 'rtl' ? 'المادة' : 'Course',           student.course])
    if (student.section)    rows.push([dir === 'rtl' ? 'الشعبة' : 'Section',          student.section])
    if (student.instructor) rows.push([dir === 'rtl' ? 'المدرس' : 'Instructor',      student.instructor])
    if (auth.user?.email)        rows.push([dir === 'rtl' ? 'البريد' : 'Email',            auth.user.email])
    if (rows.length) {
      const tableRows = rows.map(([k, v]) =>
        `<tr><td class="param-label">${eh(k)}</td><td class="param-value">${eh(v)}</td></tr>`
      ).join('')
      studentHtml = `
<div class="section">
  <h3 class="section-title">📋 ${dir === 'rtl' ? 'معلومات الطالب' : 'Student Information'}</h3>
  <table class="params-table"><tbody>${tableRows}</tbody></table>
</div>`
    }
  } catch { /* stores not ready */ }

  /* ── meta section ── */
  const metaRows = Object.entries(options.meta ?? {})
    .map(([k, v]) => `<div><span class="meta-key">${eh(k)}:</span> ${eh(v)}</div>`)
    .join('')

  /* ── params table ── */
  let paramsHtml = ''
  if (options.params && options.params.length) {
    const rows = options.params.map(p =>
      `<tr><td class="param-label">${eh(p.label)}</td><td class="param-value">${eh(String(p.value))}${p.unit ? ` <span class="unit">${eh(p.unit)}</span>` : ''}</td></tr>`
    ).join('')
    paramsHtml = `
<section class="sec">
  <h2 class="sec-heading params-heading">⚙️ ${eh(strings.paramsHeading)}</h2>
  <table class="params-table"><tbody>${rows}</tbody></table>
</section>`
  }

  /* ── summary stats cards ── */
  let statsHtml = ''
  if (options.summaryStats && options.summaryStats.length) {
    const cards = options.summaryStats.map(s =>
      `<div class="stat-card${s.highlight ? ' highlight' : ''}">
        <span class="stat-label">${eh(s.label)}</span>
        <span class="stat-value">${eh(String(s.value))}${s.unit ? `<span class="stat-unit"> ${eh(s.unit)}</span>` : ''}</span>
      </div>`
    ).join('')
    statsHtml = `
<section class="sec">
  <h2 class="sec-heading results-heading">📊 ${eh(strings.resultsHeading)}</h2>
  <div class="stats-row">${cards}</div>
</section>`
  }

  /* ── data tables ── */
  const tablesHtml = (options.tables ?? []).map(table => {
    const cap  = table.caption ? `<h3 class="table-caption">${eh(table.caption)}</h3>` : ''
    const head = `<tr>${table.headers.map(h => `<th>${eh(String(h))}</th>`).join('')}</tr>`
    const body = table.rows.map(row =>
      `<tr>${row.map(c => `<td>${eh(c === undefined ? '' : String(c))}</td>`).join('')}</tr>`
    ).join('')
    return `<section class="sec">${cap}<table><thead>${head}</thead><tbody>${body}</tbody></table></section>`
  }).join('')

  /* ── images ── */
  const allImages: LabReportImage[] = []
  if (options.canvasSnapshot) allImages.push({ src: options.canvasSnapshot, caption: strings.simulationSnap, fullWidth: true })
  allImages.push(...(options.images ?? []))
  const imagesHtml = allImages
    .filter((img) => img.src)
    .map((img, idx) => {
      const safeSrc = img.src.replace(/"/g, '&quot;')
      const cls = img.fullWidth ? 'img-full' : ''
      const breakCls = idx > 0 ? ' print-break' : ''
      return `<section class="sec${breakCls}"><p class="img-cap">${eh(img.caption ?? '')}</p><img alt="" src="${safeSrc}" class="${cls}"/></section>`
    })
    .join('')

  /* ── html blocks ── */
  const blocksHtml = (options.htmlBlocks ?? []).map(block => {
    const titleHtml = block.title ? `<h2 class="sec-heading">${eh(block.title)}</h2>` : ''
    return `<section class="sec html-block">${titleHtml}${block.html}</section>`
  }).join('')

  /* ── CSS ── */
  const isRtl = dir === 'rtl'
  const borderSide = isRtl ? 'right' : 'left'
  const _padSide   = isRtl ? 'padding-right' : 'padding-left'
  void _padSide

  const css = `
*{box-sizing:border-box;}
body{font-family:"Segoe UI",Cairo,Tajawal,sans-serif;padding:0;color:#1e293b;max-width:960px;margin:0 auto;line-height:1.6;background:#f8fafc;}
/* header */
.report-header{background:linear-gradient(135deg,#4f46e5,#0891b2);color:#fff;padding:28px 32px;border-radius:0 0 1.5rem 1.5rem;margin-bottom:24px;}
.report-header h1{margin:0;font-size:1.6rem;font-weight:800;}
.report-header .subtitle{opacity:.85;font-size:.92rem;margin-top:6px;}
/* meta */
.meta{background:#fff;border:1px solid #e2e8f0;padding:14px 18px;border-radius:1rem;font-size:.85rem;margin:0 0 16px;display:flex;flex-wrap:wrap;gap:12px 28px;box-shadow:0 1px 4px rgba(0,0,0,.06);}
.meta-key{font-weight:700;color:#475569;}
/* sections */
.sec{margin:0 0 20px;}
.sec-heading{font-size:1rem;font-weight:700;margin:0 0 10px;padding:7px 14px;border-radius:.6rem;display:flex;align-items:center;gap:8px;}
.params-heading{background:#eef2ff;color:#3730a3;border-${borderSide}:4px solid #6366f1;}
.results-heading{background:#ecfdf5;color:#065f46;border-${borderSide}:4px solid #059669;}
/* params table */
.params-table{width:100%;border-collapse:collapse;font-size:.85rem;}
.params-table td{border:1px solid #e2e8f0;padding:7px 12px;}
.param-label{font-weight:600;color:#334155;width:55%;background:#f8fafc;}
.param-value{font-family:monospace;color:#1e3a8a;}
.unit{color:#64748b;font-weight:normal;font-family:inherit;}
/* stats cards */
.stats-row{display:flex;flex-wrap:wrap;gap:10px;}
.stat-card{background:#fff;border:1px solid #e2e8f0;border-radius:1rem;padding:10px 16px;min-width:140px;flex:1;text-align:center;box-shadow:0 1px 4px rgba(0,0,0,.06);}
.stat-card.highlight{border-color:#f97316;background:#fff7ed;}
.stat-label{display:block;font-size:.75rem;font-weight:600;color:#64748b;margin-bottom:4px;}
.stat-value{display:block;font-size:1.15rem;font-weight:800;font-family:monospace;color:#1e293b;}
.stat-unit{font-size:.78rem;color:#64748b;font-weight:normal;}
/* data tables */
.table-caption{font-size:.92rem;font-weight:700;color:#334155;margin:0 0 6px;}
table{width:100%;border-collapse:collapse;margin:0;font-size:.8rem;}
th,td{border:1px solid #e2e8f0;padding:6px 8px;text-align:center;}
th{background:#e0e7ff;color:#3730a3;font-weight:700;}
tr:nth-child(even) td{background:#f8fafc;}
/* images */
img{max-width:100%;border:1px solid #e2e8f0;border-radius:.75rem;display:block;margin:8px auto;box-shadow:0 2px 8px rgba(0,0,0,.08);}
img.img-full{width:100%;}
.img-cap{text-align:center;font-size:.82rem;color:#64748b;margin:0 0 4px;}
/* html blocks */
.html-block{background:#fff;border:1px solid #e2e8f0;padding:14px 18px;border-radius:1rem;font-size:.9rem;box-shadow:0 1px 4px rgba(0,0,0,.06);}
/* footer */
.footer{margin-top:32px;text-align:center;color:#94a3b8;font-size:.75rem;border-top:1px solid #e2e8f0;padding-top:12px;}
/* actions */
.actions{display:flex;justify-content:flex-end;gap:10px;margin:20px 0;}
.btn-print{background:linear-gradient(135deg,#4f46e5,#6366f1);color:#fff;border:none;padding:10px 22px;border-radius:999px;cursor:pointer;font:inherit;font-weight:700;font-size:.92rem;}

.btn-send-teacher{background:linear-gradient(135deg,#7c3aed,#4f46e5);color:#fff;border:none;padding:10px 22px;border-radius:999px;cursor:pointer;font:inherit;font-weight:700;font-size:.92rem;box-shadow:0 4px 12px rgba(79,70,229,.3);}

  body{background:#fff;}
  .report-header{border-radius:0;}
  .stat-card,.html-block{break-inside:avoid;}

`

  const htmlContent = `<!DOCTYPE html>
<html lang="${locale}" dir="${dir}">

  <h1>${options.icon ? options.icon + ' ' : ''}${eh(options.title)}</h1>

  <div><span class="meta-key">${eh(strings.dateLabel)}:</span> ${eh(now)}</div>

  const blob = new Blob([htmlContent], { type: 'text/html; charset=utf-8' })
  const blobUrl = URL.createObjectURL(blob)
  const win = window.open(blobUrl, '_blank')
  setTimeout(() => URL.revokeObjectURL(blobUrl), 60_000)

  if (!win) return null

  if (options.openPrintDialog) {
    win.addEventListener('load', () => {
      setTimeout(() => {
        try {
          win.focus()
          win.print()
        } catch {
          /* popup blocked or print unavailable */
        }
      }, 450)
    })
  }
  return win
}

function buildSendScript(options: OpenLabReportOptions): string {
  // Serialize report data as JSON for the popup's inline JS
  const payload = {
    title: options.title,
    experimentName: options.experimentName || options.title,
    params: options.params || [],
    summaryStats: options.summaryStats || [],
    tables: options.tables || [],
    images: (options.images || []).map(img => ({ caption: img.caption || '', src: img.src.slice(0, 100) })),
    htmlBlocks: (options.htmlBlocks || []).map(b => ({ title: b.title || '', html: b.html.slice(0, 500) })),
    canvasSnapshot: options.canvasSnapshot ? '[snapshot]' : null,
    submittedAt: new Date().toISOString(),
  }
  const payloadJson = JSON.stringify(payload).replace(/"/g, '&quot;')

  return `<button id="btn-send" class="btn-send-teacher" type="button" onclick="sendToTeacher()">📤 إرسال للمدرس</button>
<script>
function sendToTeacher() {
  const btn = document.getElementById('btn-send');
  if (btn.disabled) return;

  try {
    const data = JSON.parse("${payloadJson}");
    const studentName = 'طالب ضيف';
    const classInfoRaw = localStorage.getItem('auth_classes');
    const classes = classInfoRaw ? JSON.parse(classInfoRaw) : [];
    const classInfo = classes[0] || null;
    if (!classInfo || !classInfo.code) {
      alert('يجب الانضمام لفصل أولاً');
      return;
    }

    const report = {
      id: 'guest-report-' + Date.now(),
      attemptId: Date.now(),
      experimentId: 0,
      experimentTitle: data.experimentName || data.title,
      branch: 'فيزياء',
      studentName: studentName,
      classCode: classInfo.code,
      classId: classInfo.id,
      notes: data.htmlBlocks.map(b => (b.title || '') + '\\n' + b.html).join('\\n\\n'),
      readings: data.tables.flatMap(t => t.rows.map((row, i) => {
        const obj = { '#': i + 1 };
        t.headers.forEach((h, hi) => { obj[String(h)] = row[hi] !== undefined ? String(row[hi]) : ''; });
        return obj;
      })),
      completedSteps: [],
      status: 'submitted',
      submittedAt: data.submittedAt,
    };

    const reports = JSON.parse(localStorage.getItem('physlab_guest_reports') || '[]');
    reports.push(report);
    localStorage.setItem('physlab_guest_reports', JSON.stringify(reports));

    const teacherClasses = JSON.parse(localStorage.getItem('physlab_guest_classes') || '[]');
    const cls = teacherClasses.find(c => c.code === classInfo.code);
    if (cls && cls.students) {
      const stu = cls.students.find(s => s.name === studentName);
      if (stu) {
        if (!stu.reports) stu.reports = [];
        stu.reports.push(report);
        stu.reportCount = stu.reports.length;
        localStorage.setItem('physlab_guest_classes', JSON.stringify(teacherClasses));
      }
    }

    btn.disabled = true;
    btn.className = 'btn-sent';
    btn.textContent = '✅ تم الإرسال';
    alert('تم إرسال التقرير للمدرس بنجاح!');
  } catch (e) {
    alert('خطأ: ' + e.message);
  }
}
</script>`
}

function eh(s: string): string {
  return String(s)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
}

/* ─── Experiment-state bundle (for teacher grading) ─── */

export interface ExperimentStateBundle {
  experiment: string
  branch: string
  initial_params: Record<string, unknown>
  results_array: Record<string, unknown>[]
  quiz_score?: number
  quiz_answers?: Record<string, { answer: string; correct: boolean }>
  timestamp: string
}

export function bundleLabReport(
  experiment: string,
  branch: string,
  initialParams: Record<string, unknown>,
  resultsArray: Record<string, unknown>[],
  quizScore?: number,
  quizAnswers?: Record<string, { answer: string; correct: boolean }>,
): ExperimentStateBundle {
  return {
    experiment,
    branch,
    initial_params: initialParams,
    results_array: resultsArray,
    quiz_score: quizScore,
    quiz_answers: quizAnswers,
    timestamp: new Date().toISOString(),
  }
}
